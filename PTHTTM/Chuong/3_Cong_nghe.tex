\subsection{Chức năng từng bộ phận}
\begin{itemize}
	\item Quản trị hệ thống: duy trì, bảo trì hệ thống, quản lý tài khoản người dùng.
	\item Cảnh sát: Cơ quan cảnh sát, người chịu trách nhiệm xem xét, tiếp nhận báo cáo mất tích, lập hồ sơ tìm kiếm đưa lên hệ thống để bắt đầu tìm kiếm dựa trên dữ liệu thu thập được.
	\item Người báo cáo mất tích: Người thân hoặc gia đình, có trách nhiệm cung cấp thông tin như ảnh, video, tên, tuổi, thời gian có thể mất tích, nơi nhìn thấy lần cuối cùng, … những thông tin có thể giúp ích cho việc tìm kiếm, cung cấp cho nơi lập hồ sơ mất tích như cục cảnh sát.
	\item Người tìm kiếm: Có thể là người lập hồ sơ, cảnh sát, hoặc tham gia hệ thống như một tình nguyện viên, được cung cấp thông tin người mất tích có cự ly gần với vị trí của người tìm thấy, có thể báo cáo lên hệ thống và cung cấp bằng chứng rằng có thể tìm thấy người bị mất tích.
\end{itemize}
\section{Quy trình nghiệp vụ}
\subsection{Tạo, Xác Minh và Quản Lý Hồ Sơ Người Mất Tích}
\begin{itemize}
	\item Tiếp nhận thông tin
	      \begin{itemize}
		      \item Người báo cáo mất tích tới cơ quan có thẩm quyền gần nhất để báo cáo về ca mất tích cho người lập hồ sơ
		      \item Người báo cáo mất tích cung cấp hồ sơ người bị mất tích cho người lập hồ sơ bao gồm các thông tin cơ bản như: họ và tên, tuổi, giới tính, hình ảnh hoặc video có chứa khuôn mặt của người bị mất tích, thời điểm gần nhất còn liên lạc với người bị mất tích, địa điểm lần cuối nhìn thấy người mất tích.
	      \end{itemize}
	\item Lập hồ sơ người mất tích
	      \begin{itemize}
		      \item Báo cáo người mất tích phải đảm bảo được 24 giờ sau
	      \end{itemize}
	\item Ra quyết định với hồ sơ mất tích
	      \begin{itemize}
		      \item Sau khi đã xác nhận đầy đủ thông tin, người lập hồ sơ tiến hành đưa hồ sơ người mất tích vào hệ thống.
		      \item Hồ sơ người mất tích đặt ở trạng thái ban đầu là ‘Đang tìm kiếm’
	      \end{itemize}
\end{itemize}
\subsection{Thông báo kết quả}
\begin{itemize}
	\item Nếu có bất kỳ thông báo nào đối với hồ sơ mà người dùng đang được theo dõi, người dùng sẽ được gửi thông báo kèm theo hình ảnh xác minh, thời gian, địa điểm
	\item Nếu trường hợp xác nhận là đúng, hồ sơ sẽ được chuyển về trạng thái ‘Đã tìm thấy’, đóng hồ sơ, ngoại trừ cơ quan có thẩm quyền thì những người khác sẽ không được phép thấy lại hồ sơ.
	\item Nếu không, hồ sơ vẫn tiếp tục ở trạng thái ‘Đang tìm kiếm’.
\end{itemize}
\subsection{Thêm người tìm kiếm vào hệ thống}
\begin{itemize}
	\item Người tìm kiếm đến lập hồ sơ tham gia hệ thống với cơ quan cảnh sát, người tìm kiếm cung cấp thông tin cá nhân bao gồm: họ tên, số CCCD/CMND, hình ảnh, nơi ở hiện tại, … cũng như cam kết các điều khoản trong hệ thống.
	\item Sau khi tham gia, người tìm kiếm được cấp thông tin đăng nhập vào hệ thống.
	\item Sau khi đăng nhập vào hệ thống người tìm kiếm sẽ thấy được các hồ sơ mất tích.
	\item Hồ sơ mất tích được xếp theo ưu tiên có địa điểm mất tích gần với người tìm kiếm
\end{itemize}
\subsection{Người tìm kiếm báo cáo kết quả}
\begin{itemize}
	\item Người tìm kiếm hoặc cảnh sát cung cấp bằng chứng xác thực hồ sơ mất tích bao gồm: địa điểm, thời gian, hình ảnh hoặc video có thể so khớp với dữ liệu trên hệ thống
	\item Hệ thống xác nhận kết quả và gửi cho cảnh sát
\end{itemize}
\subsection{Hệ thống báo cáo kết quả}
\begin{itemize}
	\item Cảnh sát xác nhận kết quả
	\item Nếu kết quả cảnh sát xác nhận là đúng thì gửi thông báo cho người lập hồ sơ mất tích
	\item Nếu kết quả sai, hồ sơ vẫn được để ở trạng thái 'Đang tìm kiếm'
\end{itemize}

\subsection{Đặc tả chi tiết các usecase}
\subsubsection{Đăng nhập}
\begin{itemize}
	\item \textbf{Mô tả} : Cho phép người dùng đã có tài khoản truy cập vào hệ thống bằng thông tin xác thực của mình.
	\item \textbf{Actor}: Người dùng đã đăng ký (Người báo cáo, Quản trị viên).
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đã có một tài khoản hợp lệ.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng truy cập trang đăng nhập.
			            \item Người dùng nhập Email và Mật khẩu.
			            \item Người dùng nhấn nút "Đăng nhập".
			            \item Hệ thống kiểm tra và xác thực thông tin đăng nhập với cơ sở dữ liệu.
			            \item Nếu thông tin chính xác, hệ thống cấp quyền truy cập và chuyển hướng người dùng tới giao diện chính tương ứng với vai trò của họ.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Nếu email hoặc mật khẩu không chính xác, hệ thống hiển thị thông báo: "Thông tin đăng nhập không đúng. Vui lòng thử lại."
			            \item Nếu tài khoản đã bị khóa, hệ thống hiển thị thông báo: "Tài khoản của bạn đã bị khóa."
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Người dùng được đăng nhập thành công và có một phiên làm việc mới.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}
\subsubsection{Đăng kí}
\begin{itemize}
	\item \textbf{Mô tả} : Chức năng này cho phép người dùng mới tạo một tài khoản để có thể sử dụng các tính năng của hệ thống như báo cáo, cung cấp thông tin.
	\item \textbf{Actor}: Người dùng mới (Người báo cáo, Người tìm kiếm).
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng chưa có tài khoản trên hệ thống.
			            \item Người dùng đang ở trang đăng ký.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng nhấn vào nút hoặc liên kết "Đăng ký".
			            \item Hệ thống hiển thị một biểu mẫu (form) yêu cầu các thông tin bắt buộc như: Họ tên, Email, Số điện thoại, Mật khẩu và Xác nhận mật khẩu
			            \item Người dùng điền đầy đủ và chính xác các thông tin vào biểu mẫu.
			            \item Người dùng nhấn nút "Hoàn tất đăng ký".
			            \item Hệ thống kiểm tra tính hợp lệ của dữ liệu (email đúng định dạng, mật khẩu trùng khớp, email chưa tồn tại).
			            \item Hệ thống tạo tài khoản mới và lưu thông tin vào cơ sở dữ liệu.
			            \item Hệ thống hiển thị thông báo "Đăng ký thành công" và chuyển hướng người dùng đến trang đăng nhập.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Nếu email đã tồn tại, hệ thống báo lỗi: "Email này đã được sử dụng".
			            \item Nếu mật khẩu và xác nhận mật khẩu không khớp, hệ thống báo lỗi: "Mật khẩu không khớp".
			            \item Nếu người dùng bỏ trống các trường thông tin bắt buộc, hệ thống sẽ yêu cầu nhập đầy đủ.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Một tài khoản người dùng mới được tạo thành công trong hệ thống.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}
\subsubsection{Đăng xuất}
\begin{itemize}
	\item \textbf{Mô tả} : Cho phép người dùng đang đăng nhập kết thúc phiên làm việc một cách an toàn.
	\item \textbf{Actor}: Người dùng đã đăng nhập.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đang trong một phiên đăng nhập.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng nhấn vào nút "Đăng xuất".
			            \item Hệ thống xác nhận yêu cầu và kết thúc phiên làm việc của người dùng.
			            \item Hệ thống chuyển hướng người dùng về trang chủ hoặc trang đăng nhập.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Không có.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Phiên làm việc của người dùng bị chấm dứt.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}

\subsubsection{Báo cáo mất tích}
\begin{itemize}
	\item \textbf{Mô tả} : Cho phép người dùng tạo một hồ sơ báo cáo mới về một trường hợp mất tích.
	\item \textbf{Actor}: Người báo cáo mất tích.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đã đăng nhập vào hệ thống.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng chọn chức năng "Báo cáo mất tích".
			            \item Hệ thống hiển thị biểu mẫu để nhập thông tin về người mất tích, bao gồm: Họ tên, tuổi, đặc điểm nhận dạng, hoàn cảnh mất tích, địa điểm và thời gian nhìn thấy lần cuối, hình ảnh, thông tin liên hệ của người báo cáo.
			            \item Người dùng điền đầy đủ thông tin và nhấn "Gửi báo cáo".
			            \item Hệ thống kiểm tra dữ liệu và tạo một hồ sơ mới với trạng thái "Chờ xác thực".
			            \item Hệ thống thông báo cho người dùng đã gửi báo cáo thành công và thông báo cho Cảnh sát/Quản trị viên về báo cáo mới.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Nếu thiếu thông tin bắt buộc, hệ thống sẽ yêu cầu người dùng bổ sung.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Một hồ sơ mất tích mới được tạo và chờ để được xác thực.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}

\subsubsection{Nhận thông báo}
\begin{itemize}
	\item \textbf{Mô tả} : Hệ thống tự động gửi thông báo cho người dùng về các cập nhật quan trọng liên quan đến hoạt động của họ.
	\item \textbf{Actor}: Mọi người dùng đã đăng nhập.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đã đăng nhập.
			            \item Có một sự kiện phát sinh (ví dụ: báo cáo được duyệt, có thông tin mới...).
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Hệ thống tạo một thông báo tương ứng với sự kiện.
			            \item Hệ thống hiển thị một chỉ báo (ví dụ: số trên biểu tượng chuông) trên giao diện của người dùng.
			            \item Người dùng nhấp vào biểu tượng thông báo để xem danh sách các thông báo mới.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Không có.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Người dùng được cập nhật kịp thời về các thông tin liên quan.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}

\subsubsection{Báo cáo phát hiện, Cung cấp thông tin}
\begin{itemize}
	\item \textbf{Mô tả} : Cho phép người dùng cung cấp thông tin bổ sung hoặc báo cáo đã nhìn thấy một người trong danh sách mất tích.
	\item \textbf{Actor}: Mọi người dùng.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đang xem chi tiết một hồ sơ mất tích.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng chọn chức năng "Cung cấp thông tin" hoặc "Báo cáo phát hiện".
			            \item Hệ thống hiển thị một biểu mẫu để người dùng nhập nội dung thông tin, thời gian, địa điểm, và đính kèm bằng chứng (nếu có).
			            \item Người dùng nhấn nút "Gửi thông tin".
			            \item Hệ thống lưu lại thông tin và liên kết với hồ sơ gốc, đồng thời gửi thông báo đến người tạo báo cáo và Cảnh sát/Quản trị viên.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Không có.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Thông tin mới được cập nhật vào hồ sơ và các bên liên quan nhận được thông báo.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}

\subsubsection{Tiếp nhận thông tin mất tích, Xác thực báo cáo}
\begin{itemize}
	\item \textbf{Mô tả} : Cho phép Cảnh sát hoặc Quản trị viên xem xét, kiểm tra và thay đổi trạng thái của các báo cáo do người dùng gửi lên.
	\item \textbf{Actor}: Cảnh sát, Quản trị viên.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Actor đã đăng nhập với quyền hạn phù hợp.
			            \item Có ít nhất một báo cáo đang ở trạng thái "Chờ xác thực".
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Actor truy cập vào khu vực quản lý báo cáo.
			            \item Hệ thống hiển thị danh sách các báo cáo đang chờ.
			            \item Actor chọn một báo cáo để xem chi tiết.
			            \item Sau khi kiểm tra thông tin, actor chọn hành động "Duyệt" hoặc "Từ chối".
			            \item Nếu chọn "Duyệt", hệ thống cập nhật trạng thái hồ sơ thành "Đã xác thực" và hiển thị công khai.
			            \item Nếu chọn "Từ chối", hệ thống yêu cầu nhập lý do và cập nhật trạng thái hồ sơ thành "Bị từ chối".
			            \item Hệ thống gửi thông báo kết quả cho người đã tạo báo cáo.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Không có.
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Báo cáo được duyệt và công khai, hoặc bị từ chối và không hiển thị.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}

\subsubsection{Tìm kiếm}
\begin{itemize}
	\item \textbf{Mô tả} : Cung cấp công cụ cho phép người dùng tìm kiếm các hồ sơ người mất tích đã được duyệt.
	\item \textbf{Actor}: Mọi người dùng.
	\item \textbf{Đặc tả}
	      \begin{itemize}
		      \item \textbf{Tiền điều kiện}
		            \begin{enumerate}
			            \item Người dùng đang truy cập hệ thống.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện chính}
		            \begin{enumerate}
			            \item Người dùng truy cập vào trang "Tìm kiếm".
			            \item Hệ thống hiển thị các trường và bộ lọc tìm kiếm (ví dụ: tên, khu vực, độ tuổi, giới tính).
			            \item Người dùng nhập tiêu chí tìm kiếm và nhấn nút "Tìm kiếm".
			            \item Hệ thống truy vấn và trả về danh sách các hồ sơ mất tích phù hợp với tiêu chí.
			            \item Người dùng có thể nhấp vào một hồ sơ để xem thông tin chi tiết.
		            \end{enumerate}
		      \item \textbf{Luồng sự kiện thay thế}
		            \begin{enumerate}
			            \item Nếu không có hồ sơ nào phù hợp, hệ thống sẽ hiển thị thông báo "Không tìm thấy kết quả".
		            \end{enumerate}
		      \item \textbf{Hậu điều kiện}
		            \begin{enumerate}
			            \item Người dùng xem được danh sách các hồ sơ mất tích theo yêu cầu.
		            \end{enumerate}
	      \end{itemize}
\end{itemize}
